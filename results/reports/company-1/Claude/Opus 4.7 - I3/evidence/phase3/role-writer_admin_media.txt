<script>
let deletingMediaId = null;

// ---- drag and drop upload ----
const dropZone = document.getElementById('mediaDropZone');
const bulkInput = document.getElementById('bulkFileInput');

dropZone.addEventListener('dragover', (e) => { e.preventDefault(); dropZone.classList.add('drag-over'); });
dropZone.addEventListener('dragleave', () => dropZone.classList.remove('drag-over'));
dropZone.addEventListener('drop', (e) => {
  e.preventDefault();
  dropZone.classList.remove('drag-over');
  [...e.dataTransfer.files].forEach(uploadFile);
});
bulkInput.addEventListener('change', () => {
  [...bulkInput.files].forEach(uploadFile);
});

async function uploadFile(file) {
  const queueList = document.getElementById('uploadQueueList');
  const itemId = 'upload-' + Date.now() + '-' + Math.random().toString(36).substr(2,5);
  const item = document.createElement('div');
  item.className = 'upload-queue-item';
  item.id = itemId;
  item.innerHTML = `
    <i class="bi bi-file-earmark"></i>
    <span class="uq-name">${file.name}</span>
    <span class="uq-status">Uploading…</span>
    <div class="uq-progress"><div class="uq-fill" style="width:10%"></div></div>
  `;
  queueList.appendChild(item);

  const formData = new FormData();
  formData.append('file', file);

  try {
    item.querySelector('.uq-fill').style.width = '50%';
    const resp = await fetch('/api/admin/media/upload', {
      method: 'POST', body: formData, credentials: 'same-origin',
    });
    item.querySelector('.uq-fill').style.width = '100%';
    if (resp.ok) {
      item.querySelector('.uq-status').textContent = '✓ Done';
      item.classList.add('uq-success');
      setTimeout(() => { item.remove(); location.reload(); }, 1500);
    } else {
      const err = await resp.json();
      item.querySelector('.uq-status').textContent = '✗ ' + (err.error || 'Failed');
      item.classList.add('uq-error');
    }
  } catch(e) {
    item.querySelector('.uq-status').textContent = '✗ Network error';
    item.classList.add('uq-error');
  }
}

// ---- copy URL ----
function copyMediaUrl(url, btn) {
  navigator.clipboard.writeText(url);
  const icon = btn.querySelector('i');
  icon.className = 'bi bi-check-lg';
  setTimeout(() => { icon.className = 'bi bi-clipboard'; }, 1500);
}

// ---- delete ----
function confirmDeleteMedia(id, name) {
  deletingMediaId = id;
  document.getElementById('deleteMediaName').textContent = name;
  document.getElementById('deleteMediaModal').classList.add('active');
}
function closeDeleteMediaModal() { document.getElementById('deleteMediaModal').classList.remove('active'); }

document.getElementById('confirmDeleteMediaBtn').addEventListener('click', async () => {
  if (!deletingMediaId) return;
  const btn = document.getElementById('confirmDeleteMediaBtn');
  btn.disabled = true;
  try {
    const resp = await fetch(`/api/admin/media/${deletingMediaId}`, {
      method: 'DELETE', credentials: 'same-origin',
    });
    if (resp.ok) {
      closeDeleteMediaModal();
      document.querySelector(`[data-media-id="${deletingMediaId}"]`)?.remove();
    } else {
      const d = await resp.json();
      alert(d.error || 'Delete failed');
      btn.disabled = false;
    }
  } catch(e) {
    alert('Network error');
    btn.disabled = false;
  }
});

document.getElementById('deleteMediaModal').addEventListener('click', (e) => {
  if (e.target === e.currentTarget) closeDeleteMediaModal();
});
</script>
